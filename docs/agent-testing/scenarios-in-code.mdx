The Scenario SDK is open source and runs inside your own test runner and CI. To run scenarios from the platform against your agent's HTTP endpoint instead, with no test code, see [Connect your agent](/agent-testing/connect-your-agent).

This guide will walk you through the basic setup required to run your first scenario and see the results in LangWatch.

For advanced use cases, see the [`scenario` library documentation](https://github.com/langwatch/scenario).

## 1. Installation

Install the `scenario` library in your project:

<Tabs>
  <Tab title="Python">```bash uv add langwatch-scenario ```</Tab>
  <Tab title="TypeScript">```bash npm install @langwatch/scenario ```</Tab>
</Tabs>

## 2. Configure Environment Variables

Create a `.env` file in the root of your project:

```bash title=".env"
LANGWATCH_API_KEY="your-api-key"
LANGWATCH_ENDPOINT="https://app.langwatch.ai"
```

You can find your `LANGWATCH_API_KEY` in your [LangWatch project settings](https://app.langwatch.ai/settings).

## 3. Create a Basic Scenario

Create an agent adapter that calls your agent, then run a scenario against it. For the integration patterns, see the [agent integration guide](https://langwatch.ai/scenario/agent-integration/).

<Tabs>
  <Tab title="Python">
    ```python
    import pytest
    import scenario
    import litellm

    # Configure the default model for simulations
    scenario.configure(default_model="openai/gpt-6-luna")

    @pytest.mark.agent_test
    @pytest.mark.asyncio
    async def test_vegetarian_recipe_agent():
        # 1. Create your agent adapter
        class RecipeAgent(scenario.AgentAdapter):
            async def call(self, input: scenario.AgentInput) -> scenario.AgentReturnTypes:
                return vegetarian_recipe_agent(input.messages)

        # 2. Run the scenario
        result = await scenario.run(
            name="dinner recipe request",
            description="""
                It's saturday evening, the user is very hungry and tired,
                but have no money to order out, so they are looking for a recipe.
            """,
            agents=[
                RecipeAgent(),
                scenario.UserSimulatorAgent(),
                scenario.JudgeAgent(criteria=[
                    "Agent should not ask more than two follow-up questions",
                    "Agent should generate a recipe",
                    "Recipe should include a list of ingredients",
                    "Recipe should include step-by-step cooking instructions",
                    "Recipe should be vegetarian and not include any sort of meat",
                ])
            ],
        )

        # 3. Assert the result
        assert result.success

    # Example agent implementation using litellm
    @scenario.cache()
    def vegetarian_recipe_agent(messages) -> scenario.AgentReturnTypes:
        response = litellm.completion(
            model="openai/gpt-6-luna",
            messages=[
                {
                    "role": "system",
                    "content": """
                        You are a vegetarian recipe agent.
                        Given the user request, ask AT MOST ONE follow-up question,
                        then provide a complete recipe. Keep your responses concise and focused.
                    """,
                },
                *messages,
            ],
        )
        return response.choices[0].message
    ```

  </Tab>
  <Tab title="TypeScript">
    ```typescript
    // weather.test.ts
    import { describe, it, expect } from "vitest";
    import { openai } from "@ai-sdk/openai";
    import scenario, { type AgentAdapter, AgentRole } from "@langwatch/scenario";
    import { generateText, tool } from "ai";
    import { z } from "zod";

    describe("Weather Agent", () => {
      it("should get the weather for a city", async () => {
        // 1. Define the tools your agent can use
        const getCurrentWeather = tool({
          description: "Get the current weather in a given city.",
          parameters: z.object({
            city: z.string().describe("The city to get the weather for."),
          }),
          execute: async ({ city }) => `The weather in ${city} is cloudy with a temperature of 24°C.`,
        });

        // 2. Create an adapter for your agent
        const weatherAgent: AgentAdapter = {
          role: AgentRole.AGENT,
          call: async (input) => {
            const response = await generateText({
              model: openai("gpt-6-luna"),
              system: `You are a helpful assistant that may help the user with weather information.`,
              messages: input.messages,
              tools: { get_current_weather: getCurrentWeather },
            });

            if (response.toolCalls?.length) {
              // For simplicity, we'll just return the arguments of the first tool call
              const { toolName, args } = response.toolCalls[0];
              return {
                role: "tool",
                content: [{ type: "tool-result", toolName, result: args }],
              };
            }

            return response.text;
          },
        };

        // 3. Define and run your scenario
        const result = await scenario.run({
          name: "Checking the weather",
          description: "The user asks for the weather in a specific city, and the agent should use the weather tool to find it.",
          agents: [
            weatherAgent,
            scenario.userSimulatorAgent({ model: openai("gpt-6-luna") }),
          ],
          script: [
            scenario.user("What's the weather like in Barcelona?"),
            scenario.agent(),
            // You can use inline assertions within your script
            (state) => {
              expect(state.hasToolCall("get_current_weather")).toBe(true);
            },
            scenario.succeed("Agent correctly used the weather tool."),
          ],
        });

        // 4. Assert the final result
        expect(result.success).toBe(true);
      });
    });
    ```

  </Tab>
</Tabs>

Run the test. The scenario run appears under **Agent Testing > Results** in your LangWatch project.

## 4. Grouping Your Scenarios into Test Suites and Batches

Set stable identifiers for your scenarios, test suites and batches, so LangWatch groups the runs the same way every time:

- **`id`**: A unique and stable identifier for your scenario. Without it, the id comes from the `name`, and renaming the test starts a new history.
- **`setId`**: The test suite of the scenario. It appears under **From Code** on the **Scenarios** tab, and its runs go under a run plan of the same name on the **Results** tab.
- **`batchId`**: Groups every scenario run together in one execution, for example one CI job. A CI variable such as `process.env.GITHUB_RUN_ID` is a good value.

## 5. Let the Judge Read Your Traces

When your adapter calls an agent that runs as a separate service, the judge cannot see what happens inside it. Turn on trace fetching and the judge also reads the traces your agent reports to LangWatch, one per conversation turn, before its verdict:

<Tabs>
  <Tab title="Python">
    ```python
    scenario.configure(
        default_model="openai/gpt-6-luna",
        fetch_remote_traces=True,
        trace_wait_timeout=60.0,  # seconds to wait for traces before the verdict
    )
    ```
  </Tab>
  <Tab title="TypeScript">
    ```typescript
    const result = await scenario.run({
      // ...
      fetchRemoteTraces: true,
      traceWaitTimeoutMs: 60_000,
    });
    ```
  </Tab>
</Tabs>

Spread `input.propagation_headers` (Python) or `input.propagationHeaders` (TypeScript) onto your adapter's outgoing HTTP request, so the remote service joins each turn's trace. [Linking your traces](/agent-testing/linking-your-traces) covers the mechanism, the wait behavior, and how to write trace-aware criteria.

Also check: [Evaluators on Scenarios](https://langwatch.ai/scenario/advanced/evaluators) in the Scenario documentation, for an evaluator declared in the test itself that grades every run against a golden answer, with the same mappings and the same gate as one attached on the platform.

## 6. Set Up and Clean Up Test Data

When your scenarios need data to exist first, such as a test customer or organisation, create it in a pytest fixture. Everything after `yield` runs when the test ends, even if the test fails, so the cleanup always happens.

```python
import pytest

@pytest.fixture
def seeded_customer():
    # Set up: create the data the scenario needs
    customer = create_test_customer(name="Test Customer", plan="pro")
    created_order_ids = []
    customer["created_order_ids"] = created_order_ids

    # your test appends ids the agent creates to created_order_ids
    yield customer

    # Clean up: delete only what this run created, even if the test failed.
    # Try every delete, orders before the customer, then report every failure.
    errors = []
    for order_id in created_order_ids:
        try:
            delete_order(order_id)
        except Exception as error:
            errors.append(error)
    try:
        delete_test_customer(customer["id"])
    except Exception as error:
        errors.append(error)
    if errors:
        raise RuntimeError("cleanup failed: " + "; ".join(repr(e) for e in errors)) from errors[0]
```

`create_test_customer`, `delete_order` and `delete_test_customer` are your own helpers, for example calls to your API or database.

Pass the seeded ids to your agent, and tell the judge about the seeded state with `additional_context` on the `scenario.judge()` script step. The judge reads it under `<additional_context>`, so your criteria can name the real facts.

```python
import pytest
import scenario

@pytest.mark.agent_test
@pytest.mark.asyncio
async def test_support_agent_knows_the_customer(seeded_customer):
    class SupportAgent(scenario.AgentAdapter):
        async def call(self, input: scenario.AgentInput) -> scenario.AgentReturnTypes:
            return support_agent(
                input.messages,
                customer_id=seeded_customer["id"],
            )

    result = await scenario.run(
        name="customer asks about their plan",
        description="The customer asks which plan they are on.",
        agents=[
            SupportAgent(),
            scenario.UserSimulatorAgent(),
            scenario.JudgeAgent(criteria=[
                "Agent should state the customer's plan correctly",
            ]),
        ],
        script=[
            scenario.user(),
            scenario.agent(),
            scenario.judge(
                additional_context=(
                    f"Seeded state: customer {seeded_customer['id']} "
                    f"is on the {seeded_customer['plan']} plan."
                ),
            ),
        ],
    )

    assert result.success
```

### One test per region

To run the same scenario in several regions, seed once with a session-scoped fixture and use `pytest.mark.parametrize` for one test per region. The fixture deletes only what the run created, and only once, after the last test.

```python
@pytest.fixture(scope="session")
def seeded_regions(request):
    customers = {}
    for region in ["eu", "us"]:
        customer = create_test_customer(name="Test Customer", region=region)
        # Register cleanup right after each create, so a later failure still cleans up
        request.addfinalizer(lambda id=customer["id"]: delete_test_customer(id))
        customers[region] = customer

    return customers

@pytest.mark.asyncio
@pytest.mark.parametrize("region", ["eu", "us"])
async def test_support_agent_per_region(seeded_regions, region):
    customer = seeded_regions[region]
    ...  # build the adapter and run the scenario as above
```

If your agent is connected to LangWatch and you run scenarios from the platform, use [setup and teardown hooks](/agent-testing/setup-and-teardown-hooks) instead.
